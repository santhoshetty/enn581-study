\hd{Configuration space, inflation \pg{12--24}}
\textbf{Task space} \pg{12}: frame of the task. \textbf{C-space} $\mathcal C$ \pg{14}: generalised coordinates (joint values; $x,y,\theta$); a configuration $\mathbf q$ is a point, the robot is a point there. $\mathcal C_{free}=\{\mathbf q\in\mathcal C\mid\mathbf q\cap\mathcal C_{obs}=\varnothing\}$ \pg{15} (slide writes $\in$; intended $\subset$). Planning = a curve in $\mathcal C_{free}$. Rotary joints wrap ($q\equiv q+2\pi$): RR arm $=$ torus.
\slidefig[0.85\linewidth]{w12-lec}{19}{163 108 134 146}{2R arm: task space (left), C-space $(\theta_1,\theta_2)$ (right); a point is two C-space points (elbow up/down)}
\begin{fbx}[C-space image of a point $(x,y)$ \added]
\begin{gather*}\cos\theta_2=\frac{x^2+y^2-L_1^2-L_2^2}{2L_1L_2},\ \ \theta_2=\pm\arccos(\cdot)\\ \theta_1=\atantwo(y,x)-\atantwo(L_2\sin\theta_2,L_1+L_2\cos\theta_2)\end{gather*}
\use{$|\cos\theta_2|>1$: out of reach, no C-space point.}
\end{fbx}
\begin{worked}[$L_1=L_2=1$, $P=(1,1)$]
$\cos\theta_2=0\Rightarrow\theta_2=\pm90^\circ$; $\theta_1=45^\circ\mp45^\circ=0^\circ$ or $90^\circ$: points $(0^\circ,90^\circ)$, $(90^\circ,-90^\circ)$; FK of each gives $(1,1)$. \chk
\end{worked}
\textbf{Inflate} \pg{20--24}: disc robot of radius $r$ $\to$ point; obstacle grows by $r$ (Minkowski sum). Corridor width $w$: free width $w-2r<0\Rightarrow$ blocked.
\begin{fbx}[Grown polygon area \added]
\[A_{grown}=A+P\,r+\pi r^2\]
\use{$A$ area, $P$ perimeter of the convex polygon, $r$ robot radius.}
\end{fbx}
Triangle $(5,3),(6,3),(5,1)$, $r=0.5$: $A=1$, $P=5.236$, $A_{grown}=1+2.618+0.785=4.403$ m$^2$. \chk

\hd{BFS and DFS \pg{50--54}}\label{w12:bfs}
\begin{recipe}[BFS]
\begin{enumerate}
\item Queue $=[\text{start}]$, visited $=\{\text{start}\}$.
\item Pop the \emph{front} $n$; stop if goal.
\item Each unvisited neighbour $m$ (alphabetical): parent$(m)=n$, push to the \emph{back}, mark visited.
\item Path = parents from goal back. \textbf{Check:} fewest edges; costs ignored. Time $O(|V|+|E|)$ \pg{65}.
\end{enumerate}
\end{recipe}

\begin{recipe}[DFS (explicit stack)]
\begin{enumerate}
\item Mark visited when \emph{popped}; push neighbours in reverse alphabetical order. Stack $=[\text{start}]$.
\item Pop the \emph{top}; skip if visited; else mark, record parent (the pusher).
\item Stop if goal; else push unvisited neighbours. \textbf{Check:} \emph{a} path, not shortest; depth limit \pg{54}.
\end{enumerate}
\end{recipe}

\hd{Dijkstra \pg{55--58}}\label{w12:dij}
\begin{fbx}[Relaxation \pg{56}]
\[\text{if }g(n)+d(n,m)<g(m):\ \ g(m)\leftarrow g(n)+d(n,m),\ \ \text{parent}(m)\leftarrow n\]
\use{$g$ = cheapest known cost start$\to$node ($\infty$ if unreached); $n$ = unexpanded node with smallest $g$; $m$ its neighbour. Weights $\ge0$.}
\end{fbx}
\begin{recipe}[Dijkstra]
\begin{enumerate}
\item $g(\text{start})=0$, others $\infty$. 2. Expand smallest-$g$ unexpanded $n$ (alphabetical ties).
\item Relax each non-expanded neighbour (strict $<$); never change an expanded node.
\item \textbf{Stop when the goal is expanded}, not discovered. \textbf{Check:} $g(\text{goal})=$ sum of path edges; expansions in non-decreasing $g$.
\end{enumerate}
\end{recipe}

\hd{A* \pg{59--62}}\label{w12:astar}
\begin{fbx}[A* priority \pg{59}]
\begin{gather*}f(n)=g(n)+h(n)\\ \text{admissible: }h(n)\le d(n,\text{goal})\ \forall n\\ \text{consistent: }h(n)\le d(n,m)+h(m)\ \forall\text{ adjacent }n,m\end{gather*}
\use{$h$ = optimistic cost to go. Dijkstra with priority $g\to g+h$. Admissible $\Rightarrow$ optimal; consistent $\Rightarrow$ admissible, popped $f$ non-decreasing. $h=0$ is Dijkstra.}
\end{fbx}
\begin{recipe}[A*]
\begin{enumerate}
\item Write $h$ at each node; check admissible and consistent (every edge, both directions).
\item Pop smallest $f$ (ties: smaller $h$, then alphabetical). Relax neighbours; $f(m)=g(m)+h(m)$ uses the \emph{child's} $h$.
\item Stop when the goal is popped. \textbf{Check:} $f(\text{goal})=g(\text{goal})=$ path cost.
\end{enumerate}
\end{recipe}

\hd{Sampling-based planners \pg{66--83}}
Idea \pg{66,69}: no explicit $\mathcal C_{free}$; sampler + distance function + local planner + collision detector. PRM (1996), RRT (1998) \pg{68}.
\begin{recipe}[PRM \pg{70--72}]
\begin{enumerate}
\item Learning: sample $\mathbf q$ uniformly, discard if in $\mathcal C_{obs}$ (point-in-polygon: ray to $+x$, odd crossings = inside); add to $V$.
\item Connect to vertices within $r$ (or $k$ nearest) if the segment is free (segment--edge test).
\item Query: add start, goal; A*/Dijkstra on the roadmap. \textbf{Check:} valid, rarely shortest; disconnected $\Rightarrow$ sample more.
\end{enumerate}
\end{recipe}
\begin{fbx}[Hit probability \added]
\[p=\frac{A_{goal}}{A_{\mathcal C}},\quad E[N]=\frac1p,\quad P(\ge1\text{ hit in }N)=1-(1-p)^N\]
\use{practical map: $p=\pi/100=0.0314$, $E[N]=31.8$.}
\end{fbx}

\hd{Choosing a planner \pg{80--83}}\label{w12:which}
\emph{Complete}: finds a path if one exists, else reports failure. \emph{Resolution complete}: at the chosen grid resolution. \emph{Probabilistically complete}: $P(\text{found})\to1$ as samples $\to\infty$; may not terminate if no path \pg{81}.\par
\scriptsize\setlength{\tabcolsep}{2pt}
\begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash}p{0.17\linewidth}>{\raggedright\arraybackslash}p{0.25\linewidth}>{\raggedright\arraybackslash}p{0.22\linewidth}>{\raggedright\arraybackslash}X@{}}\toprule
Planner & Complete? & Optimal? & Use when\\\midrule
BFS & resolution (finite graph) & fewest edges; cost only if equal costs & equal moves\\
DFS & not on deep graphs & no & any path, low memory\\
Dijkstra & resolution & cheapest, costs $\ge0$ & costs, no goal estimate\\
A* & resolution & cheapest if $h$ admissible & costs + distance $h$; 2D grids\\
Visibility & complete & shortest & 2D polygons (inflate)\\
Voronoi & complete & max clearance & range-sensor following\\
PRM & probabilistic & no & static map, many queries, high DoF\\
RRT & probabilistic & no & single query, differential constraints\\\bottomrule
\end{tabularx}
\normalsize
\begin{recipe}[``Which planner?'' \added]
\begin{enumerate}
\item $d\le3$ and known map: grid/visibility graph; large $d$: grid needs $N^d$ nodes $\Rightarrow$ sample.
\item Car/diff-drive: RRT with sampled controls. Static map, many queries: PRM; one query: RRT.
\item Costs + good guess: A*; costs only: Dijkstra; equal costs: BFS. Name the method \emph{and} its guarantee.
\end{enumerate}
\end{recipe}
