\hd{1. Vectors}\nopagebreak[4]
\begin{fbx}[Dot, norm, angle, projection \added]
\[\mathbf a\!\cdot\!\mathbf b=\mathbf a^\top\mathbf b=a_1b_1+a_2b_2+a_3b_3=|\mathbf a||\mathbf b|\cos\varphi,\quad|\mathbf a|=\sqrt{\mathbf a\!\cdot\!\mathbf a}\]
\[\hat{\mathbf a}=\tfrac{\mathbf a}{|\mathbf a|},\quad\varphi=\arccos\tfrac{\mathbf a\cdot\mathbf b}{|\mathbf a||\mathbf b|}\in[0,\pi],\quad\operatorname{proj}_{\mathbf b}\mathbf a=\tfrac{\mathbf a\cdot\mathbf b}{\mathbf b\cdot\mathbf b}\mathbf b\]
$\mathbf a\!\cdot\!\mathbf b=0\iff\perp$ (columns of $\Rm$ are unit and $\perp$).
\end{fbx}
\begin{fbx}[Cross product: determinant (mind the $\hat{\mathbf j}$ sign) \added]
\[\mathbf a\times\mathbf b=\begin{vmatrix}\hat{\mathbf i}&\hat{\mathbf j}&\hat{\mathbf k}\\a_1&a_2&a_3\\b_1&b_2&b_3\end{vmatrix}
=\begin{bmatrix}a_2b_3-a_3b_2\\ a_3b_1-a_1b_3\\ a_1b_2-a_2b_1\end{bmatrix}\ \ (\hat{\mathbf j}\text{ term has a }{-})\]
\[\mathbf a\times\mathbf b=[\mathbf a]_\times\mathbf b,\quad[\mathbf a]_\times=\begin{bmatrix}0&-a_3&a_2\\a_3&0&-a_1\\-a_2&a_1&0\end{bmatrix},\quad\mathbf a=\begin{bmatrix}S_{32}\\S_{13}\\S_{21}\end{bmatrix}\ (\text{vex})\]
$|\mathbf a\times\mathbf b|=|\mathbf a||\mathbf b|\sin\varphi$ = parallelogram area; direction by right-hand rule; $\mathbf a\times\mathbf b=-\mathbf b\times\mathbf a$; $\mathbf a\times\mathbf a=\mathbf0$. Skew $\mathbf S=[\mathbf a]_\times$: $\mathbf S^\top=-\mathbf S$, $\mathbf S\mathbf a=\mathbf 0$, $\det\mathbf S=0$, $\mathbf S^2=\mathbf a\mathbf a^\top-|\mathbf a|^2\mathbf I$, $\Rm[\mathbf a]_\times\Rm^\top=[\Rm\mathbf a]_\times$.\\
$\mathbf a\!\cdot\!(\mathbf b\times\mathbf c)=\det[\mathbf a^\top;\mathbf b^\top;\mathbf c^\top]$ = signed volume (0 if coplanar) $=\mathbf b\!\cdot\!(\mathbf c\times\mathbf a)$; $\mathbf a\times(\mathbf b\times\mathbf c)=\mathbf b(\mathbf a\!\cdot\!\mathbf c)-\mathbf c(\mathbf a\!\cdot\!\mathbf b)$.\\
Trap check: $(1,2,2)\times(2,0,1)=(2,3,-4)$. \chk
\end{fbx}

\hd{2. Matrices}\nopagebreak[4]
\begin{fbx}[$2\times2$ and $3\times3$: det, inverse, eigenvalues \added]
\[\mathbf A=\begin{bmatrix}a&b\\c&d\end{bmatrix},\ \det\mathbf A=ad-bc,\ \mathbf A^{-1}=\tfrac1{ad-bc}\begin{bmatrix}d&-b\\-c&a\end{bmatrix}\]
\[\lambda^2-(a+d)\lambda+(ad-bc)=0,\ \ \lambda_{1,2}=\tfrac{(a+d)\pm\sqrt{(a+d)^2-4(ad-bc)}}2,\ \ \mathbf v=\begin{bmatrix}b\\\lambda-a\end{bmatrix}\]
$\lambda_1+\lambda_2=a+d$, $\lambda_1\lambda_2=ad-bc$; eigenvectors of a symmetric $\mathbf A$ are $\perp$.
\[\det\begin{bmatrix}a&b&c\\d&e&f\\g&h&i\end{bmatrix}=a(ei-fh)-b(di-fg)+c(dh-eg)\]
\[\mathbf A^{-1}=\tfrac1{\det\mathbf A}\operatorname{adj}\mathbf A,\quad\operatorname{adj}\mathbf A=\begin{bmatrix}ei-fh&ch-bi&bf-ce\\fg-di&ai-cg&cd-af\\dh-eg&bg-ah&ae-bd\end{bmatrix}\]
$(\mathbf{AB})^\top=\mathbf B^\top\mathbf A^\top$, $(\mathbf{AB})^{-1}=\mathbf B^{-1}\mathbf A^{-1}$ (order reverses). $\det=0\iff$ singular, rank drops. Check: $\begin{bmatrix}2&1\\1&3\end{bmatrix}$: $\lambda=(5\pm\sqrt5)/2$. \chk
\end{fbx}
\begin{fbx}[Positive definite test; pseudo-inverse \added]
$\mathbf M=\mathbf M^\top\succ0\iff\mathbf x^\top\mathbf M\mathbf x>0\ \forall\mathbf x\ne\mathbf 0\iff$ all $\lambda>0\iff$ all leading minors $>0$ ($M_{11}>0$, $M_{11}M_{22}-M_{12}^2>0$, \dots). E.g. $\begin{bmatrix}1&2\\2&1\end{bmatrix}$: $\det=-3$, $\lambda=3,-1$, indefinite. \chk
\[\mathbf J\,(m\!\times\!n,\ \text{rank }m<n):\ \mathbf J^{+}=\mathbf J^\top(\mathbf J\mathbf J^\top)^{-1},\ \ \dot{\mathbf q}=\mathbf J^{+}\boldsymbol\nu\Rightarrow\mathbf J\dot{\mathbf q}=\boldsymbol\nu,\ |\dot{\mathbf q}|\text{ minimal}\]
Tall $\mathbf J$ (rank $n$): $\mathbf J^{+}=(\mathbf J^\top\mathbf J)^{-1}\mathbf J^\top$ (least squares). Used W6 (redundant arm), W10 (extra wheels). Check: $\mathbf J=\begin{bmatrix}1&0&1\\0&1&1\end{bmatrix}$, $\boldsymbol\nu=(1,2)$: $\dot{\mathbf q}=(0,1,1)$, $|\dot{\mathbf q}|=\sqrt2$. \chk
\end{fbx}

\begin{fbx}[Basic rotations (right-hand rule) \pg{W2 10--12}]
\hd{3. Rotations and homogeneous transforms}
{\small\[\Rm_x=\begin{bmatrix}1&0&0\\0&c\theta&-s\theta\\0&s\theta&c\theta\end{bmatrix}\ \Rm_y=\begin{bmatrix}c\theta&0&s\theta\\0&1&0\\-s\theta&0&c\theta\end{bmatrix}\ \Rm_z=\begin{bmatrix}c\theta&-s\theta&0\\s\theta&c\theta&0\\0&0&1\end{bmatrix}\]}
$\Rm^\top\Rm=\mathbf I$, $\Rm^{-1}=\Rm^\top$, $\det\Rm=+1$, ${}^A\mathbf P={}^A\Rm_B{}^B\mathbf P$; columns of ${}^A\Rm_B$ = $\hat x_B,\hat y_B,\hat z_B$ in $\{A\}$. $\theta>0$ anticlockwise looking down the axis to the origin. Same-axis angles add; different axes do not commute. Minus sign: upper right in $\Rm_x,\Rm_z$ but lower left in $\Rm_y$.
\end{fbx}
\begin{twoway}{Current (moving) axes}{Fixed (world) axes}
Turn $\Rm_1$ then $\Rm_2$ about the \emph{new} axes: $\Rm=\Rm_1\Rm_2$ (written order = done order; ETS, DH).
\tcblower
Turn $\Rm_1$ then $\Rm_2$ about the \emph{original} axes: $\Rm=\Rm_2\Rm_1$ (last written = last done).
\end{twoway}
Check: $\Rm_z(90^\circ)\Rm_x(90^\circ)=\begin{bmatrix}0&0&1\\1&0&0\\0&1&0\end{bmatrix}\ne\Rm_x(90^\circ)\Rm_z(90^\circ)=\begin{bmatrix}0&-1&0\\0&0&-1\\1&0&0\end{bmatrix}$. \chk
\begin{fbx}[Rodrigues; angles from $\Rm$ ($r_{ij}$ = row $i$, col $j$) \pg{W3 38}]
\[\Rm(\hat{\mathbf n},\theta)=\mathbf I+s\theta\,[\hat{\mathbf n}]_\times+(1-c\theta)[\hat{\mathbf n}]_\times^2,\qquad\boldsymbol\omega=\dot\theta\hat{\mathbf n}\]
\textbf{ZYZ} $\Rm_z(\alpha)\Rm_y(\beta)\Rm_z(\gamma)$: $r_{13}=c_\alpha s_\beta,\ r_{23}=s_\alpha s_\beta,\ r_{33}=c_\beta,\ r_{31}=-s_\beta c_\gamma,\ r_{32}=s_\beta s_\gamma$
\[\beta=\atantwo\!\big(\sqrt{r_{13}^2+r_{23}^2},r_{33}\big),\ \alpha=\atantwo(r_{23},r_{13}),\ \gamma=\atantwo(r_{32},-r_{31})\]
2nd solution: $-\beta,\ \alpha+\pi,\ \gamma+\pi$. If $s_\beta=0$ only $\alpha\pm\gamma$ is fixed. Check $(30^\circ,40^\circ,50^\circ)$ round-trips. \chk\\
\textbf{RPY} $\Rm_z(\psi)\Rm_y(\theta)\Rm_x(\phi)$: $r_{11}=c_\psi c_\theta,\ r_{21}=s_\psi c_\theta,\ r_{31}=-s_\theta,\ r_{32}=c_\theta s_\phi,\ r_{33}=c_\theta c_\phi$
\[\theta=\atantwo\!\big(-r_{31},\sqrt{r_{11}^2+r_{21}^2}\big),\ \psi=\atantwo(r_{21},r_{11}),\ \phi=\atantwo(r_{32},r_{33})\]
2nd solution: $\pi-\theta,\ \psi+\pi,\ \phi+\pi$; singular at $\theta=\pm90^\circ$.
\end{fbx}
\begin{fbx}[Pose $\T$, inverse, composition, elementary transforms \pg{W2 12--16}]
\[{}^A\T_B=\begin{bmatrix}{}^A\Rm_B&{}^A\mathbf P_B\\\mathbf 0^\top&1\end{bmatrix},\ \ \begin{bmatrix}{}^A\mathbf P\\1\end{bmatrix}={}^A\T_B\begin{bmatrix}{}^B\mathbf P\\1\end{bmatrix},\ \ {}^A\T_C={}^A\T_B{}^B\T_C\]
\[({}^A\T_B)^{-1}={}^B\T_A=\begin{bmatrix}{}^A\Rm_B^\top&-{}^A\Rm_B^\top\,{}^A\mathbf P_B\\\mathbf 0^\top&1\end{bmatrix}\]
\[t(a,b,c)=\begin{bmatrix}1&0&0&a\\0&1&0&b\\0&0&1&c\\0&0&0&1\end{bmatrix},\quad\Rm_{x,y,z}(\theta)\text{ as }4\times4=\begin{bmatrix}\Rm&\mathbf 0\\\mathbf 0^\top&1\end{bmatrix}\]
ETS = product read left to right, each step about the \emph{current} axes. Proximal DH: ${}^{i-1}\T_i=\Rm_x(\alpha_{i-1})t_x(a_{i-1})\Rm_z(\theta_i)t_z(d_i)$. A direction uses $\Rm$ only (4th entry 0). Order trap: $t(2,1,0)\Rm_z(90^\circ)$ has position $(2,1,0)$; $\Rm_z(90^\circ)t(2,1,0)$ has $(-1,2,0)$. Check: $\Rm_z(90^\circ)$, $\mathbf P=(2,1,0)$ maps $(1,0,0)\mapsto(2,2,0)$; inverse translation $(-1,2,0)$. \chk
\end{fbx}

\begin{fbx}[$\atantwo$; $a\cos\theta+b\sin\theta=c$; half-angle; quadratic]
$\theta=\atantwo(y,x)\in(-\pi,\pi]$, \textbf{$y$ first}. $\arctan(y/x)$ cannot tell $(1,1)$ from $(-1,-1)$.
\begin{tabularx}{\linewidth}{@{}ccccX@{}}\toprule

$x,y$ & $+,\ge0$ & $-,\ge0$ & $-,-$ & $+,-$\\

\end{tabularx}
$x=0$: $\pm\pi/2$. Check: $(-1,1)\to135^\circ$ (not $-45^\circ$); $(-1,-1)\to-135^\circ$. \chk
\[r=\sqrt{a^2+b^2},\ \varphi=\atantwo(b,a):\ \ a\cos\theta+b\sin\theta=r\cos(\theta-\varphi)=c\]
\[\Rightarrow\ \theta=\varphi\pm\arccos\tfrac cr\]

\begin{fbx}[Chain rule, product rule, $\dot\Rm$, Lagrange partials \pg{W6 18--19, W7 15--17}]
\hd{5. Calculus of joint angles}
\[\tfrac d{dt}f(q_1,q_2)=\textstyle\sum_i\frac{\partial f}{\partial q_i}\dot q_i,\qquad\tfrac d{dt}(uv)=\dot uv+u\dot v\]
\[\begin{array}{ll}\frac d{dt}\cos q_1=-\sin q_1\,\dot q_1&\frac d{dt}\sin q_1=\cos q_1\,\dot q_1\\[2pt]
\frac d{dt}c_{12}=-s_{12}(\dot q_1+\dot q_2)&\frac d{dt}s_{12}=c_{12}(\dot q_1+\dot q_2)\\[2pt]
\frac d{dt}\dot q^2=2\dot q\ddot q&\frac d{dt}(\cos q\,\dot q)=-\sin q\,\dot q^2+\cos q\,\ddot q\end{array}\]
Inner rate: $\dot q_1+\dot q_2$ in $c_{12}$. RR tip: $\dot x=-L_1s_1\dot q_1-L_2s_{12}(\dot q_1+\dot q_2)$, $\dot y=L_1c_1\dot q_1+L_2c_{12}(\dot q_1+\dot q_2)$; $L_i=1$, $\mathbf q=(0,90^\circ)$, $\dot{\mathbf q}=(1,0)$: $(\dot x,\dot y)=(-1,1)=\boldsymbol\omega\times\mathbf r$. \chk
\[\dot\Rm=[\boldsymbol\omega]_\times\Rm,\quad[\boldsymbol\omega]_\times=\dot\Rm\Rm^\top,\quad\tfrac d{d\theta}\Rm_z=\begin{bmatrix}-s\theta&-c\theta&0\\c\theta&-s\theta&0\\0&0&0\end{bmatrix}\]
\[\mathcal L=K-P,\quad Q_i=\tfrac d{dt}\tfrac{\partial\mathcal L}{\partial\dot q_i}-\tfrac{\partial\mathcal L}{\partial q_i}\]
Treat $q_i,\dot q_i$ as independent symbols for $\partial$; then differentiate $\partial\mathcal L/\partial\dot q_i$ in time. $K=\frac12\dot{\mathbf q}^\top\mathbf M\dot{\mathbf q}\Rightarrow\partial K/\partial\dot{\mathbf q}=\mathbf M\dot{\mathbf q}$. $Q_i$: N\,m (revolute), N (prismatic). \end{fbx}
